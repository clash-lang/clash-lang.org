{{- /* Inline an SVG that lives in the static directory matching this page's URL (static/blog/-/), so its CSS participates in the page cascade and can respond to the site's data-theme dark mode toggle. Usage: inline-svg "figure.svg" */}} {{- $name := .Get 0 -}} {{- $file := printf "static%s%s" .Page.RelPermalink $name -}} {{- $svg := readFile $file -}} {{- if hasSuffix $name ".drawio.svg" -}} {{- /* draw.io renders every text label as a holding an HTML followed by a base64 PNG fallback for renderers without foreignObject support. Every browser takes the foreignObject branch when the SVG is inlined in HTML, so the PNGs (typically over half of the file) are dead weight: drop them. Only the trailing fallback inside a is removed, so images deliberately embedded in a diagram are kept. */ -}} {{- $svg = replaceRE `]*href="data:image/[^"]*"[^>]*/>\s*` "" $svg -}} {{- /* The root element's `content` attribute holds the diagram source (an escaped mxfile) so draw.io can re-open the exported file. The file on disk keeps it; the inlined copy does not need it. */ -}} {{- $svg = replaceRE `(]*?) content="[^"]*"` "$1" $svg -}} {{- end -}} {{- $svg | safeHTML -}}
