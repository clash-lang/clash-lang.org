{{- /* Inline an SVG that lives in the static directory matching this page's URL (static/blog/-/), so its CSS participates in the page cascade and can respond to the site's data-theme dark mode toggle. Usage: inline-svg "figure.svg" inline-svg "figure.svg" "A caption, with **markdown** support." A caption is rendered LaTeX style: centered under the figure and automatically prefixed with "Figure N:" (numbered per page, captioned figures only; see static/css/inline-svg.css). Files named *.drawio.svg get automatic dark mode support and responsive sizing (see static/css/inline-svg.css). For this to work they must be exported from draw.io with adaptive colors: check "Appearance: Automatic" in the File > Export as > SVG dialog. Such exports pick their colors with the CSS light-dark() function, which inline-svg.css pins to the site theme. */}} {{- $name := .Get 0 -}} {{- $file := printf "static%s%s" .Page.RelPermalink $name -}} {{- $svg := readFile $file -}} {{- /* Drop the XML prolog, DOCTYPE, and comments preceding the root element; they are not valid when inlined into HTML. */ -}} {{- $svg = replaceRE `(?s)^.*? holding an HTML followed by a base64 PNG fallback for renderers without foreignObject support. Every browser takes the foreignObject branch when the SVG is inlined in HTML, so the PNGs (typically over half of the file) are dead weight: drop them. Only the trailing fallback inside a is removed, so images deliberately embedded in a diagram are kept. */ -}} {{- $svg = replaceRE `]*href="data:image/[^"]*"[^>]*/>\s*` "" $svg -}} {{- /* The root element's `content` attribute holds the diagram source (an escaped mxfile) so draw.io can re-open the exported file. The file on disk keeps it; the inlined copy does not need it. */ -}} {{- $svg = replaceRE `(]*?) content="[^"]*"` "$1" $svg -}} {{- end -}} {{- /* NB: the class is not called "inline-svg", because the theme already uses that name for its menu icons. */ -}}
